If the workload truly needs higher privileges

            

              Developers cannot self-escalate. A cluster-admin must bind the ServiceAccount to an elevated SCC.
            

            

              

                1. CLI grant
                

                  
oc adm policy add-scc-to-user anyuid -z default -n prod-workloads

                  
                

              

              

                2. GitOps RoleBinding
                

                  
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: system:openshift:scc:anyuid
  namespace: prod-workloads
subjects:
- kind: ServiceAccount
  name: default
  namespace: prod-workloads
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: system:openshift:scc:anyuid
